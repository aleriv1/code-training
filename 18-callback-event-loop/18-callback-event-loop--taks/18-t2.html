function runCode() {
          console.log("before promise"); // 1 Синхрон
          return new Promise((resolve) => {
            setTimeout(() => {
              console.log("Zero Promise"); //4 макротаск из callback queue от setTimout; 
              // resolve внутри, потому microtask после здесь

              resolve();
            }, 0);
          });
        }
        
        setTimeout(() => {
          console.log("Zero"); // 3 макротаск из callback queue
        }, 0);
        
        runCode().then(() => console.log("Zero Promise Invoked")); // 5 Микротаск, 
        // но в теле коллбека таймаута
        
        console.log("One"); // 2 синхрон
  1. before promise
  2. One
  3. Zero
  4. Zero Promise
  5. Zero Promise Invoked
